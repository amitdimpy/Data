---
# Purpose: To calculate the disk space consumption and usage in percentage with RG,SubsID
# Scope: Subscriptions

- hosts: Subcription1:Subcription2
  gather_facts: no

  tasks:

  - name: Disk usage and Disk free
    shell: |
      hname=`uname -n`
      dir=/home/snbadmin/calc_stg;dirp=$dir/disksize;dirl=$dir/log
      fname=$dir/$hname.disk_details.txt
      tmpf=$dirl/t0.txt;tmpf1=$dirl/t1.txt;tmpf2=$dirl/t2.txt;tmpf3=$dirl/t3.txt;tmpf4=$dirl/t4.txt;tmpf5=$dirl/t5.txt;tmpf6=$dirl/t6.txt
      tmpf7=$dirl/t7.txt;tmpf8=$dirl/t8.txt;tmpf9=$dirl/t9.txt;tmpf10=$dirl/t10.txt;tmpf11=$dirl/t11.txt;tmpf12=$dirl/t12.txt
      tmpf13=$dirl/t13.txt;tmpf14=$dirl/t14.txt;tmpf15=$dirl/t15.txt;tmpf16=$dirl/t16.txt;tmpf17=$dirl/t17.txt;tmpf18=$dirl/t18.txt
      tmpf19=$dirl/t19.txt;tmpf20=$dirl/t20.txt;tmpf21=$dirl/t21.txt;tmpf22=$dirl/t22.txt;tmpf23=$dirl/t23.txt;tmpf24=$dirl/t24.txt
      tmpf25=$dirl/t25.txt;tmpf26=$dirl/t26.txt;tmpf27=$dirl/t27.txt;tmpf28=$dirl/t28.txt;tmpf29=$dirl/t29.txt;tmpf30=$dirl/t30.txt
      tmpf15_1=$dirl/t15_1.txt;tmpf15_2=$dirl/t15_2.txt;tmpf15_3=$dirl/t15_3.txt
      log_date=`date +%d-%h-%Y-%H-%M-%S`
      rm $tmpf $tmpf1 $tmpf2 $tmpf3 $tmpf4 $tmpf5 $tmpf6 $tmpf7 $tmpf8 $tmpf9 $tmpf10 $tmpf11 $tmpf12 $tmpf13 $tmpf14 $tmpf15 > /dev/null 2>&1
      rm $tmpf16 $tmpf17 $tmpf18 $tmpf19 $tmpf20 $tmpf21 $tmpf22 $tmpf23 $tmpf24 $tmpf25 $tmpf26 $tmpf27 $tmpf28 $tmpf29 $tmpf30 > /dev/null 2>&1
      rm $tmpf15_1 $tmpf15_2 $tmpf15_3 > /dev/null 2>&1
      if [ ! -d "$dir" ]
      then
        mkdir -p $dir 
      fi
      if [ ! -d "$dirp" ]
      then
        mkdir -p $dirp 
      fi
      if [ ! -d "$dirl" ]
      then
        mkdir -p $dirl
      fi
      if [ ! -f "$fname" ]
      then
      	echo "File with disk details does not exist"
      	exit 1
      fi
      sed -e 's/"//g' -e 's/,//g' $fname | tr -d "[:blank:]" > $tmpf   #Format data
      grep -iwE 'lun|diskSizeGB|storageAccountType|id' $tmpf | grep -vE 'networkProfile|settings|extensions|proximity|galleries|virtualMachines|key:disk.lun|networkInterfaces' > $tmpf1     #Get specific column data
      for num in `nl $tmpf1  | grep "lun:" | awk '{print $1}'`
      do
        a=$num;b=`expr $num + 3`
        sed -n "$a,$b"p $tmpf1 >> $tmpf2
      done
      grep "lun:" $tmpf2 | sed 's/\s$//g' | awk -F":" '{print $NF}' > $tmpf3    #LUN ID from API data
      grep "storageAccountType:" $tmpf2 | sed 's/\s$//g' | awk -F":" '{print $NF}' > $tmpf4   #Storage Account Type from API data
      grep "id:" $tmpf2 | sed 's/\s$//g' | awk -F"/" '{print $3}' > $tmpf5   #Subscription ID from API data
      grep "id:" $tmpf2 | sed 's/\s$//g' | awk -F"/" '{print $5}' > $tmpf6   #Resource Group Name from API data
      grep "id:" $tmpf2 | sed 's/\s$//g' | awk -F"/" '{print $NF}' > $tmpf7  #Disk Name from API data 
      grep "diskSizeGB:" $tmpf2  | sed 's/\s$//g' | awk -F":" '{print $NF}' > $tmpf8   #Disk Size from API data
      paste $tmpf3 $tmpf4 $tmpf5 $tmpf6 $tmpf7 $tmpf8 > $tmpf9    #Paste data in one file       
      ls -l /dev/disk/azure/scsi1 | sed 's# -> ../../../#==#g' | grep -vE 'total|-part' | awk '{print $NF}' | sed -e 's/lun//g' -e 's#==#==/dev/#g' -e 's/==/ /g' > $tmpf10  #Lun id with disk name
      pvs > $tmpf11 2>&1; grep -vE 'Couldn|WARNING' $tmpf11 | sed 's/<//g' | awk 'BEGIN{FS=OFS=" "} {gsub(/m/, "/1024", $NF)} 1' | awk 'BEGIN{FS=OFS=" "} {gsub(/t/, "*1024", $NF)} 1' | awk 'BEGIN{FS=OFS=" "} {gsub(/g/, "*1", $NF)} 1' | grep -ivE 'root|free' | awk '{print $1,$2,$NF}' > $tmpf12 #Capture pvs output
      vgs > $tmpf13 2>&1; grep -vE 'Couldn|WARNING' $tmpf13 | sed 's/<//g' | awk 'BEGIN{FS=OFS=" "} {gsub(/m/, "/1024", $NF)} 1' | awk 'BEGIN{FS=OFS=" "} {gsub(/t/, "*1024", $NF)} 1' | awk 'BEGIN{FS=OFS=" "} {gsub(/g/, "*1", $NF)} 1' | grep -ivE 'root|free' | awk '{print $1,$6,$NF}' > $tmpf14 #Capture vgs output
      for chk in `awk '{print $NF}' $tmpf10`
      do
        lnid=`grep -w $chk $tmpf10 | awk '{print $1}'`
        dskid=`grep -w $chk $tmpf12`
        echo $lnid $dskid >> $tmpf15     #Lun id and disk details from lsscsi and pvs output
      done
      grep -vn vg $tmpf15 | awk -F":" '{print $1}' > $tmpf15_1
      grep -vn vg $tmpf15 | awk -F":" '{print $2}' > $tmpf15_2
      for ln1 in `cat $tmpf15_1`
      do
       for inp1 in `cat $tmpf15_2`
       do
        sed -i ''$ln1's/'$inp1'/'$inp1' Unused_or_VxVM_Disk NA 0/' $tmpf15
       done
      done
      awk '{print $1,$2,$3,$4}' $tmpf15 > $tmpf15_3
      cp -p $tmpf15_3 $tmpf15        
      for calc in `awk '{print $NF}' $tmpf15`
      do
       ans=`echo $calc | bc`
       echo $ans | sed 's/\s$//' >> $tmpf16     #Calculation of PV free size in GB
      done
      awk '{print $1,$2,$3}' $tmpf15 | sed 's/\s$//' > $tmpf17   #First,second,third column of combined output of lsscsi,pvs
      paste $tmpf17 $tmpf16 > $tmpf18   #Paste calculated disk size with disk lun id,vgname of disk 
      sort -k1 $tmpf18 | awk '{print $4,$3,$2}' | sed 's/\s$//' > $tmpf19    #Sort disk size on basis of lun id
      sort -k1 $tmpf9 | sed 's/\s$//' > $tmpf20     #Sort subsc,rg,storage type,disk name on basis of lun id
      paste $tmpf20 $tmpf19 > $tmpf21    #Output with all details of rg,disk size,free disk,subs id 
      awk 'BEGIN { OFS = "\t" } { $10 = $6 - $7 } 1' $tmpf21 > $tmpf22    #Column with used capacity added
      awk 'BEGIN { OFS = "\t" } { $11 = ((($6 - $7)/($6))*100) } 1' $tmpf22 > $tmpf23   #Column of used percent
      awk 'BEGIN { OFS = "\t" } { $12 = ((($6 - $10)/($6))*100) } 1' $tmpf23 > $tmpf24  #Column of free percent 
      echo "VGName DiskSize(GB) Used_Space(GB) Free_Space(GB) %DiskUsed %DiskFree DiskName StgActType DeviceName Res_Grp Subs_Id" > $tmpf25
      awk '{print $8,$6,$10,$7,$11,$12,$9,$2,$5,$4,$3}' $tmpf24 >> $tmpf25    #Arrange data header wise
      sed 's/ /,/g;2,$ s#^#'$hname',#g;s/VGName/ServerName,VGName/g;2,$ s#$#,'$log_date'#g' $tmpf25 > $dir/disksize/$hname.diskanalysis.csv  #Replace space with commas and add host name in first line
      sed '1d' $dir/disksize/$hname.diskanalysis.csv > $dir/disksize/$hname.diskanalysis1.csv
      grep -v osdisk $dir/disksize/$hname.diskanalysis1.csv > $dir/disksize/$hname.diskanalysis.csv
#      if [ ! -s "${tmpf12}" ]
#      then
#        echo "$hname,No_disk_pvs_output,No_VG,NA,NA,NA,NA,NA,NA,NA,NA,NA,$log_date" > $dir/disksize/$hname.diskanalysis.csv  #No disk in pvs
#      fi

  - name: Fetch files
    fetch: 
      src: "/home/snbadmin/calc_stg/disksize/{{ inventory_hostname }}.diskanalysis.csv"
      dest: "/tmp/amitsh/playbooks/AS/AZURE/diskinfo/"
      flat: yes

  - name: Mail and append
    shell: |
      cd /tmp/amitsh/playbooks/AS/AZURE/diskinfo/
      for i in `ls -lrt | grep "{{ inventory_hostname }}" | grep diskanalysis | awk '{print $NF}'`
      do
          cat $i >> prod_allvms_output.csv
      done
    delegate_to: localhost	 

  - lineinfile:
      path: /tmp/amitsh/playbooks/AS/AZURE/diskinfo/prod_allvms_output.csv
      insertbefore: BOF
      line: "ServerName,VGName,DiskSize(GB),Used_Space(GB),Free_Space(GB),%DiskUsed,%DiskFree,DiskName,StgActType,DeviceName,Res_Grpr,Subs_Id,Execution_Date"
    delegate_to: localhost
