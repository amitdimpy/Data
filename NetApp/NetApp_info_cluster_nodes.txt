def node_detail(paswd):
    res1=requests.get('https://gpxclus01.sgp.st.com/api/cluster/nodes?vendor_serial_number=*&system_id=*&version.full=*&model=FAS8200&state=up&ha.enabled=true&serial_number=*&location=*&fields=uptime&return_records=true&return_timeout=15', auth = ('discvadm' , paswd), verify=False, timeout=5)
    file1 = open("gpxclus01_details.txt","a")
    print('NODE_NAME',"",'SERIAL_NO',"   ",'MODEL'," ",'SYSTEM_ID',"                      ",'ONTAP_VERSION',"                                                ",'UPTIME',"  ",'HA_STATUS', file=file1)
    print('----------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------', file=file1)
    for data in res1.json()['records']:
        print(data['name'],"   ",data['serial_number'],data['model'],data['system_id'],"       ",data['version']['full'],"  ",data['uptime'],"  ",data['ha']['enabled'], file=file1)
    print("\n", file=file1)
    file1.close();
